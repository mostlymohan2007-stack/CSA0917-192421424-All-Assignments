CLASS Hostel IMPLEMENTS Serializable   { serialVersionUID <- 1L;  String name;  List<Room> rooms }
CLASS Room   IMPLEMENTS Serializable   { serialVersionUID <- 1L;  int number;  List<Resident> residents }
CLASS Resident IMPLEMENTS Serializable {
    serialVersionUID <- 1L
    String  name;  String regNo;                    // persisted
    TRANSIENT String     password;                  // secret: must never be written to a file
    TRANSIENT Connection connection;                // live resource: not serializable, meaningless in another run
    METHOD readObject(stream): stream.defaultReadObject();  connection <- NULL     // re-acquired lazily after loading
}
ALGORITHM SaveHostel(h, file):   OPEN ObjectOutputStream(file);  WRITE h  // writes the WHOLE reachable graph;  CLOSE
ALGORITHM LoadHostel(file):      OPEN ObjectInputStream(file);   h <- READ;  CLOSE;  RETURN h
